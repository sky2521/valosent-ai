@app.route('/api/analyze', methods=['POST'])
def api_analyze_review():
    """Primary asynchronous endpoint for multi-tier sentiment inference."""
    payload = request.get_json(force=True, silent=True)
    if not payload or 'text' not in payload:
        return jsonify({"error": "Invalid request: 'text' field is required"}), 400
        
    raw_text = str(payload['text']).strip()
    if len(raw_text) > 2000:
        return jsonify({"error": "Payload exceeds 2,000 character limit"}), 413
        
    # Execute tiered inference cascade
    result = analyze_sentiment_cascade(raw_text)
    
    # Asynchronously log review to local database
    try:
        new_entry = ReviewHistory(
            review_text=raw_text,
            predicted_label=result['label'],
            confidence_score=result['confidence'],
            model_tier=result['tier'],
            timestamp=datetime.utcnow()
        )

        db.session.commit()
    except Exception as db_err:
        db.session.rollback()
        app.logger.warning(f"Local logging failed: {db_err}")
        
    return jsonify(result), 200

\section*{Listing A.2: Hybrid Sentiment Engine \& XAI Attributor (\texttt{src/sentiment.py})}
\addcontentsline{toc}{section}{Listing A.2: Hybrid Sentiment Engine & XAI Attributor (src/sentiment.py)}

\begin{lstlisting}[language=Python, caption={Multi-tier hybrid sentiment coordinator and XAI token saliency engine (\texttt{src/sentiment.py}).}]
import numpy as np
from typing import Dict, Any, List
from src.text_processing import clean_gaming_text
from src.ml_model import get_cached_linear_svc_pipeline
from src.inference import get_cached_distilbert_engine

def compute_xai_saliency(cleaned_text: str, model_pipeline) -> List[Dict[str, Any]]:
    """Computes token-level linear feature attribution for UI highlight rendering."""
    tokens = cleaned_text.split()
    vectorizer = model_pipeline.named_steps['vectorizer']
    classifier = model_pipeline.named_steps['classifier']
    
    feature_names = vectorizer.get_feature_names_out()
    vocab_map = {word: idx for idx, word in enumerate(feature_names)}
    
    # Extract linear hyperplane coefficients
    if hasattr(classifier, 'calibrated_classifiers_'):
        coefs = np.mean([clf.estimator.coef_ for clf in classifier.calibrated_classifiers_], axis=0)
    else:
        coefs = classifier.coef_
        
    # Class 0: Negative, Class 1: Neutral, Class 2: Positive
    pos_weights = coefs[2] if coefs.shape[0] > 1 else coefs[0]
    neg_weights = coefs[0]
    
    token_saliency = []
    for token in tokens:
        idx = vocab_map.get(token)
        if idx is not None:
            weight = float(pos_weights[idx] - neg_weights[idx])
        else:
            weight = 0.0
            
        token_saliency.append({
            "token": token,
            "weight": round(weight, 4),
            "saliency_type": "positive" if weight > 0.05 else ("negative" if weight < -0.05 else "neutral")
        })
    return token_saliency

def analyze_sentiment_cascade(raw_text: str, tau_1: float = 0.80) -> Dict[str, Any]:
    """Executes the adaptive multi-tier cascading inference protocol."""
    cleaned = clean_gaming_text(raw_text)
    svc_pipeline = get_cached_linear_svc_pipeline()
    
    # Tier 1 Execution (LinearSVC)
    svc_probs = svc_pipeline.predict_proba([cleaned])[0]
    classes = svc_pipeline.classes_
    top_idx = int(np.argmax(svc_probs))
    top_conf = float(svc_probs[top_idx])
    
    # If Tier-1 confidence satisfies boundary tau_1, return immediately
    if top_conf >= tau_1:
        saliency = compute_xai_saliency(cleaned, svc_pipeline)
        return {
            "tier": "Tier-1: LinearSVC",
            "label": classes[top_idx],
            "confidence": round(top_conf, 4),
            "latency_tier": "fast",
            "saliency": saliency
        }
        
    # Tier 3 Execution (DistilBERT Contextual Transformer)
    transformer = get_cached_distilbert_engine()
    bert_res = transformer.predict(cleaned)
    bert_res["tier"] = "Tier-3: DistilBERT"
    bert_res["latency_tier"] = "deep_context"
    bert_res["saliency"] = compute_xai_saliency(cleaned, svc_pipeline)
    
    return bert_res
\end{lstlisting}

\section*{Listing A.4: Cloud PostgreSQL Sync Interface (\texttt{src/supabase\_backend.py})}
\addcontentsline{toc}{section}{Listing A.4: Cloud PostgreSQL Sync Interface (src/supabase_backend.py)}

\begin{lstlisting}[language=Python, caption={Supabase PostgreSQL cloud synchronization client (\texttt{src/supabase_backend.py}).}]
import os
import requests
from typing import Dict, Any

SUPABASE_URL = os.getenv("SUPABASE_URL", "")
SUPABASE_KEY = os.getenv("SUPABASE_KEY", "")

def sync_review_to_supabase(review_payload: Dict[str, Any]) -> bool:
    """Asynchronously dispatches evaluated review telemetry to remote PostgreSQL."""
    if not SUPABASE_URL or not SUPABASE_KEY:
        return False
        
    endpoint = f"{SUPABASE_URL}/rest/v1/reviews_audit"
    headers = {
        "apikey": SUPABASE_KEY,
        "Authorization": f"Bearer {SUPABASE_KEY}",
        "Content-Type": "application/json",
        "Prefer": "return=minimal"
    }
    
    try:
        response = requests.post(endpoint, json=review_payload, headers=headers, timeout=3.0)
        return response.status_code in [200, 201]
    except requests.RequestException:
        return False
\end{lstlisting}
